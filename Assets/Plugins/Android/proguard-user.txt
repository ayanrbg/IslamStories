# Custom R8/ProGuard rules, applied on top of Unity's proguard-unity.txt and the
# consumer rules that ship inside the Firebase / Play Services AARs.
#
# Everything here exists for one reason: R8 only sees Java call sites. Anything
# reached by *name* — from C# via AndroidJavaClass, from the manifest, or over JNI —
# looks unused to it and gets stripped or renamed. Those entry points are listed below.

# -- Unity player ------------------------------------------------------------
# PushService.cs:212 resolves "com.unity3d.player.UnityPlayer" by string and reads
# its static currentActivity field. proguard-unity.txt already keeps this; repeated
# here so the dependency stays visible if Unity ever trims its own rules.
-keep class com.unity3d.player.** { *; }

# -- Firebase Messaging activity ---------------------------------------------
# Generated by FirebaseMessagingActivityGenerator and wired in through the merged
# manifest. AGP normally derives a keep rule from the manifest, but the class is
# also swapped in at build time, so pin it explicitly.
-keep class com.google.firebase.MessagingUnityPlayerActivity { *; }

# -- Firebase / Play Services ------------------------------------------------
# The Unity wrapper libraries (firebase-app-unity, firebase-*-unity) call back into
# managed code and are themselves invoked reflectively by the C# layer.
-keep class com.google.firebase.** { *; }
-keep class com.google.android.gms.** { *; }
-dontwarn com.google.firebase.**
-dontwarn com.google.android.gms.**

# flatbuffers is used for Firebase's serialized message payloads; its generated
# classes are instantiated reflectively.
-keep class com.google.flatbuffers.** { *; }

# -- Google Play Billing (Unity IAP) -----------------------------------------
# Unity IAP 5.x ships no Java layer of its own: it reaches the billing library
# entirely from C# by string name (Runtime/Purchasing/Stores/Android/GooglePlay/AAR/*.cs)
# and implements the listener interfaces through AndroidJavaProxy, which dispatches
# by method name. R8 therefore sees zero Java call sites and shrinks the library to
# almost nothing — in the first minified build BillingResult had already become
# "com.android.billingclient.api.a" and purchases were dead.
# Members matter as much as class names here: renaming a proxied interface method
# breaks dispatch just as thoroughly as stripping the class.
# The dependency is injected at build time by the package's IAPAndroidDependencies.cs
# (com.android.billingclient:billing:8.0.0), which is why it never appears in
# mainTemplate.gradle's resolver block.
-keep class com.android.billingclient.** { *; }
-keep interface com.android.billingclient.** { *; }
-keep class com.unity.purchasing.** { *; }
-keep interface com.unity.purchasing.** { *; }
-dontwarn com.android.billingclient.**

# -- JNI ---------------------------------------------------------------------
# Native methods must keep their exact names or the JNI lookup fails at runtime.
-keepclasseswithmembernames class * {
    native <methods>;
}

# -- Reflection metadata -----------------------------------------------------
# Signature is required for generic types (Firebase's Task<T> chains);
# the annotation attributes keep Firebase's own keep-by-annotation rules working.
-keepattributes Signature, *Annotation*, InnerClasses, EnclosingMethod, Exceptions

# -- Crash reports -----------------------------------------------------------
# Without this, obfuscated Java stack traces lose line numbers entirely. The
# SourceFile rename keeps the mapping honest while staying obfuscated.
-keepattributes SourceFile, LineNumberTable
-renamesourcefileattribute SourceFile
